## Introduction

An in-memory cache (or key-value store) is a fundamental component in system design, used to reduce latency and load on slower storage systems (databases, disk, network). Examples include Redis, Memcached, and application-level caches.

### Key Design Goals

| Goal | Description |
|------|-------------|
| **Fast Access** | O(1) average-case lookup, insert, delete |
| **Bounded Memory** | Never exceed configured memory limit |
| **Smart Eviction** | Remove least valuable entries when full |
| **Thread Safety** | Support concurrent access from multiple threads |
| **TTL Support** | Automatic expiration of stale entries |

## Data Structures

The choice of data structure directly impacts cache performance.

### Hash Table - The Foundation

For O(1) key-based access, a hash table is essential.

```cpp
#include <functional>
#include <vector>
#include <list>
#include <optional>

template<typename K, typename V>
class BasicHashTable {
private:
    struct Entry {
        K key;
        V value;
        bool occupied = false;
    };
    
    std::vector<std::list<Entry>> buckets_;
    size_t size_ = 0;
    size_t bucket_count_;
    float max_load_factor_ = 0.75f;
    std::hash<K> hasher_;
    
public:
    explicit BasicHashTable(size_t initial_buckets = 16)
        : buckets_(initial_buckets), bucket_count_(initial_buckets) {}
    
    void put(const K& key, const V& value) {
        if (loadFactor() > max_load_factor_) {
            rehash(bucket_count_ * 2);
        }
        
        size_t idx = hasher_(key) % bucket_count_;
        
        // Check if key exists
        for (auto& entry : buckets_[idx]) {
            if (entry.key == key) {
                entry.value = value;
                return;
            }
        }
        
        // Insert new entry
        buckets_[idx].push_back({key, value, true});
        size_++;
    }
    
    std::optional<V> get(const K& key) {
        size_t idx = hasher_(key) % bucket_count_;
        
        for (const auto& entry : buckets_[idx]) {
            if (entry.key == key) {
                return entry.value;
            }
        }
        
        return std::nullopt;
    }
    
    bool remove(const K& key) {
        size_t idx = hasher_(key) % bucket_count_;
        
        auto& bucket = buckets_[idx];
        for (auto it = bucket.begin(); it != bucket.end(); ++it) {
            if (it->key == key) {
                bucket.erase(it);
                size_--;
                return true;
            }
        }
        
        return false;
    }
    
    size_t size() const { return size_; }
    
private:
    float loadFactor() const {
        return static_cast<float>(size_) / bucket_count_;
    }
    
    void rehash(size_t new_bucket_count) {
        std::vector<std::list<Entry>> new_buckets(new_bucket_count);
        
        for (auto& bucket : buckets_) {
            for (auto& entry : bucket) {
                size_t new_idx = hasher_(entry.key) % new_bucket_count;
                new_buckets[new_idx].push_back(std::move(entry));
            }
        }
        
        buckets_ = std::move(new_buckets);
        bucket_count_ = new_bucket_count;
    }
};
```

### Robin Hood Hashing

An improved open addressing technique that reduces probe length variance:

```cpp
template<typename K, typename V>
class RobinHoodHashTable {
private:
    struct Entry {
        K key;
        V value;
        uint32_t probe_distance = 0;  // Distance from ideal position
        bool occupied = false;
    };
    
    std::vector<Entry> table_;
    size_t size_ = 0;
    size_t capacity_;
    std::hash<K> hasher_;
    
public:
    explicit RobinHoodHashTable(size_t capacity = 16)
        : table_(capacity), capacity_(capacity) {}
    
    void put(const K& key, const V& value) {
        if (size_ >= capacity_ * 0.7) {
            resize(capacity_ * 2);
        }
        
        Entry new_entry{key, value, 0, true};
        size_t idx = hasher_(key) % capacity_;
        
        while (true) {
            if (!table_[idx].occupied) {
                // Empty slot found
                table_[idx] = std::move(new_entry);
                size_++;
                return;
            }
            
            if (table_[idx].key == key) {
                // Update existing
                table_[idx].value = value;
                return;
            }
            
            // Robin Hood: steal from the rich (longer probe) 
            // to give to the poor (shorter probe)
            if (new_entry.probe_distance > table_[idx].probe_distance) {
                std::swap(new_entry, table_[idx]);
            }
            
            new_entry.probe_distance++;
            idx = (idx + 1) % capacity_;
        }
    }
    
    std::optional<V> get(const K& key) {
        size_t idx = hasher_(key) % capacity_;
        uint32_t probe = 0;
        
        while (table_[idx].occupied && probe <= table_[idx].probe_distance) {
            if (table_[idx].key == key) {
                return table_[idx].value;
            }
            probe++;
            idx = (idx + 1) % capacity_;
        }
        
        return std::nullopt;
    }
    
private:
    void resize(size_t new_capacity) {
        std::vector<Entry> old_table = std::move(table_);
        table_ = std::vector<Entry>(new_capacity);
        capacity_ = new_capacity;
        size_ = 0;
        
        for (auto& entry : old_table) {
            if (entry.occupied) {
                put(entry.key, entry.value);
            }
        }
    }
};
```

### Doubly-Linked List for LRU

To track access order efficiently:

```cpp
template<typename K, typename V>
struct LRUNode {
    K key;
    V value;
    LRUNode* prev = nullptr;
    LRUNode* next = nullptr;
    
    LRUNode() = default;
    LRUNode(K k, V v) : key(std::move(k)), value(std::move(v)) {}
};

template<typename K, typename V>
class DoublyLinkedList {
private:
    LRUNode<K, V> head_;  // Dummy head (MRU end)
    LRUNode<K, V> tail_;  // Dummy tail (LRU end)
    size_t size_ = 0;
    
public:
    DoublyLinkedList() {
        head_.next = &tail_;
        tail_.prev = &head_;
    }
    
    // Add to front (most recently used)
    void pushFront(LRUNode<K, V>* node) {
        node->prev = &head_;
        node->next = head_.next;
        head_.next->prev = node;
        head_.next = node;
        size_++;
    }
    
    // Remove from anywhere
    void remove(LRUNode<K, V>* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
        size_--;
    }
    
    // Move existing node to front
    void moveToFront(LRUNode<K, V>* node) {
        remove(node);
        pushFront(node);
    }
    
    // Get least recently used (from back)
    LRUNode<K, V>* back() {
        if (size_ == 0) return nullptr;
        return tail_.prev;
    }
    
    // Remove and return least recently used
    LRUNode<K, V>* popBack() {
        if (size_ == 0) return nullptr;
        LRUNode<K, V>* node = tail_.prev;
        remove(node);
        return node;
    }
    
    size_t size() const { return size_; }
    bool empty() const { return size_ == 0; }
};
```

## Eviction Policies

When the cache is full, we need to decide which entry to remove.

### LRU (Least Recently Used)

The most common policy. Evicts the entry that hasn't been accessed for the longest time.

```cpp
template<typename K, typename V>
class LRUCache {
private:
    struct Node {
        K key;
        V value;
        Node* prev = nullptr;
        Node* next = nullptr;
    };
    
    std::unordered_map<K, Node*> cache_;
    Node head_, tail_;  // Dummy nodes
    size_t capacity_;
    size_t size_ = 0;
    
    void moveToHead(Node* node) {
        removeNode(node);
        addToHead(node);
    }
    
    void removeNode(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }
    
    void addToHead(Node* node) {
        node->prev = &head_;
        node->next = head_.next;
        head_.next->prev = node;
        head_.next = node;
    }
    
    Node* removeTail() {
        Node* node = tail_.prev;
        removeNode(node);
        return node;
    }
    
public:
    explicit LRUCache(size_t capacity) : capacity_(capacity) {
        head_.next = &tail_;
        tail_.prev = &head_;
    }
    
    ~LRUCache() {
        for (auto& [key, node] : cache_) {
            delete node;
        }
    }
    
    std::optional<V> get(const K& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return std::nullopt;  // Cache miss
        }
        
        // Move to front (most recently used)
        moveToHead(it->second);
        return it->second->value;
    }
    
    void put(const K& key, const V& value) {
        auto it = cache_.find(key);
        
        if (it != cache_.end()) {
            // Update existing entry
            it->second->value = value;
            moveToHead(it->second);
            return;
        }
        
        // Create new entry
        Node* node = new Node{key, value};
        cache_[key] = node;
        addToHead(node);
        size_++;
        
        // Evict if over capacity
        if (size_ > capacity_) {
            Node* lru = removeTail();
            cache_.erase(lru->key);
            delete lru;
            size_--;
        }
    }
    
    bool remove(const K& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return false;
        }
        
        removeNode(it->second);
        delete it->second;
        cache_.erase(it);
        size_--;
        return true;
    }
    
    size_t size() const { return size_; }
};
```

### LFU (Least Frequently Used)

Evicts the entry with the lowest access count.

```cpp
template<typename K, typename V>
class LFUCache {
private:
    struct Node {
        K key;
        V value;
        size_t frequency = 1;
        std::list<K>::iterator list_iter;
    };
    
    size_t capacity_;
    size_t min_frequency_ = 0;
    
    std::unordered_map<K, Node> cache_;
    std::unordered_map<size_t, std::list<K>> frequency_lists_;
    
public:
    explicit LFUCache(size_t capacity) : capacity_(capacity) {}
    
    std::optional<V> get(const K& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return std::nullopt;
        }
        
        updateFrequency(it->second);
        return it->second.value;
    }
    
    void put(const K& key, const V& value) {
        if (capacity_ == 0) return;
        
        auto it = cache_.find(key);
        
        if (it != cache_.end()) {
            it->second.value = value;
            updateFrequency(it->second);
            return;
        }
        
        // Evict if necessary
        if (cache_.size() >= capacity_) {
            evictLFU();
        }
        
        // Insert new entry with frequency 1
        auto& freq_list = frequency_lists_[1];
        freq_list.push_front(key);
        
        cache_[key] = Node{key, value, 1, freq_list.begin()};
        min_frequency_ = 1;
    }
    
private:
    void updateFrequency(Node& node) {
        size_t old_freq = node.frequency;
        size_t new_freq = old_freq + 1;
        
        // Remove from old frequency list
        frequency_lists_[old_freq].erase(node.list_iter);
        
        // Update min_frequency if needed
        if (frequency_lists_[old_freq].empty()) {
            frequency_lists_.erase(old_freq);
            if (min_frequency_ == old_freq) {
                min_frequency_ = new_freq;
            }
        }
        
        // Add to new frequency list
        frequency_lists_[new_freq].push_front(node.key);
        node.list_iter = frequency_lists_[new_freq].begin();
        node.frequency = new_freq;
    }
    
    void evictLFU() {
        auto& lfu_list = frequency_lists_[min_frequency_];
        K key_to_evict = lfu_list.back();
        lfu_list.pop_back();
        
        if (lfu_list.empty()) {
            frequency_lists_.erase(min_frequency_);
        }
        
        cache_.erase(key_to_evict);
    }
};
```

### FIFO (First In, First Out)

Simple queue-based eviction:

```cpp
template<typename K, typename V>
class FIFOCache {
private:
    std::unordered_map<K, V> cache_;
    std::queue<K> order_;
    size_t capacity_;
    
public:
    explicit FIFOCache(size_t capacity) : capacity_(capacity) {}
    
    std::optional<V> get(const K& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return std::nullopt;
        }
        return it->second;
    }
    
    void put(const K& key, const V& value) {
        if (cache_.find(key) != cache_.end()) {
            cache_[key] = value;
            return;
        }
        
        while (cache_.size() >= capacity_) {
            K oldest = order_.front();
            order_.pop();
            cache_.erase(oldest);
        }
        
        cache_[key] = value;
        order_.push(key);
    }
};
```

### TTL (Time To Live)

Entries expire after a configured duration:

```cpp
template<typename K, typename V>
class TTLCache {
private:
    struct Entry {
        V value;
        std::chrono::steady_clock::time_point expiry;
    };
    
    std::unordered_map<K, Entry> cache_;
    std::chrono::milliseconds default_ttl_;
    
    // Min-heap for expiration times
    using ExpiryPair = std::pair<std::chrono::steady_clock::time_point, K>;
    std::priority_queue<ExpiryPair, std::vector<ExpiryPair>, std::greater<>> expiry_queue_;
    
public:
    explicit TTLCache(std::chrono::milliseconds ttl = std::chrono::minutes(5))
        : default_ttl_(ttl) {}
    
    void put(const K& key, const V& value, 
             std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        auto expiry = std::chrono::steady_clock::now() + 
                      ttl.value_or(default_ttl_);
        
        cache_[key] = Entry{value, expiry};
        expiry_queue_.push({expiry, key});
    }
    
    std::optional<V> get(const K& key) {
        cleanupExpired();
        
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return std::nullopt;
        }
        
        if (isExpired(it->second)) {
            cache_.erase(it);
            return std::nullopt;
        }
        
        return it->second.value;
    }
    
    void cleanupExpired() {
        auto now = std::chrono::steady_clock::now();
        
        while (!expiry_queue_.empty() && expiry_queue_.top().first <= now) {
            K key = expiry_queue_.top().second;
            expiry_queue_.pop();
            
            auto it = cache_.find(key);
            if (it != cache_.end() && it->second.expiry <= now) {
                cache_.erase(it);
            }
        }
    }
    
private:
    bool isExpired(const Entry& entry) {
        return std::chrono::steady_clock::now() >= entry.expiry;
    }
};
```

## Thread Safety

Caches are typically accessed by multiple threads concurrently.

### Approach 1: Coarse-Grained Locking

Simple but can be a bottleneck:

```cpp
template<typename K, typename V>
class ThreadSafeLRUCache {
private:
    LRUCache<K, V> cache_;
    mutable std::shared_mutex mutex_;
    
public:
    explicit ThreadSafeLRUCache(size_t capacity) : cache_(capacity) {}
    
    std::optional<V> get(const K& key) {
        // Exclusive lock because get() modifies LRU order
        std::unique_lock<std::shared_mutex> lock(mutex_);
        return cache_.get(key);
    }
    
    void put(const K& key, const V& value) {
        std::unique_lock<std::shared_mutex> lock(mutex_);
        cache_.put(key, value);
    }
    
    bool remove(const K& key) {
        std::unique_lock<std::shared_mutex> lock(mutex_);
        return cache_.remove(key);
    }
};
```

### Approach 2: Striped Locking (Sharding)

Divide the cache into independent shards:

```cpp
template<typename K, typename V, size_t SHARD_COUNT = 16>
class ShardedCache {
private:
    struct Shard {
        LRUCache<K, V> cache;
        mutable std::shared_mutex mutex;
        
        explicit Shard(size_t capacity) : cache(capacity) {}
    };
    
    std::array<std::unique_ptr<Shard>, SHARD_COUNT> shards_;
    std::hash<K> hasher_;
    
    Shard& getShard(const K& key) {
        size_t idx = hasher_(key) % SHARD_COUNT;
        return *shards_[idx];
    }
    
    const Shard& getShard(const K& key) const {
        size_t idx = hasher_(key) % SHARD_COUNT;
        return *shards_[idx];
    }
    
public:
    explicit ShardedCache(size_t total_capacity) {
        size_t per_shard = (total_capacity + SHARD_COUNT - 1) / SHARD_COUNT;
        
        for (auto& shard : shards_) {
            shard = std::make_unique<Shard>(per_shard);
        }
    }
    
    std::optional<V> get(const K& key) {
        Shard& shard = getShard(key);
        std::unique_lock<std::shared_mutex> lock(shard.mutex);
        return shard.cache.get(key);
    }
    
    void put(const K& key, const V& value) {
        Shard& shard = getShard(key);
        std::unique_lock<std::shared_mutex> lock(shard.mutex);
        shard.cache.put(key, value);
    }
    
    bool remove(const K& key) {
        Shard& shard = getShard(key);
        std::unique_lock<std::shared_mutex> lock(shard.mutex);
        return shard.cache.remove(key);
    }
    
    size_t size() const {
        size_t total = 0;
        for (const auto& shard : shards_) {
            std::shared_lock<std::shared_mutex> lock(shard->mutex);
            total += shard->cache.size();
        }
        return total;
    }
};
```

### Approach 3: Lock-Free Cache

Using atomic operations for maximum concurrency:

```cpp
template<typename K, typename V>
class LockFreeCache {
private:
    struct Node {
        K key;
        std::atomic<V> value;
        std::atomic<Node*> next;
        std::atomic<bool> valid{true};
        
        Node(K k, V v) : key(std::move(k)), value(v), next(nullptr) {}
    };
    
    static constexpr size_t BUCKET_COUNT = 1024;
    std::array<std::atomic<Node*>, BUCKET_COUNT> buckets_;
    std::atomic<size_t> size_{0};
    std::hash<K> hasher_;
    
public:
    LockFreeCache() {
        for (auto& bucket : buckets_) {
            bucket.store(nullptr, std::memory_order_relaxed);
        }
    }
    
    ~LockFreeCache() {
        for (auto& bucket : buckets_) {
            Node* node = bucket.load(std::memory_order_relaxed);
            while (node) {
                Node* next = node->next.load(std::memory_order_relaxed);
                delete node;
                node = next;
            }
        }
    }
    
    std::optional<V> get(const K& key) {
        size_t idx = hasher_(key) % BUCKET_COUNT;
        Node* node = buckets_[idx].load(std::memory_order_acquire);
        
        while (node) {
            if (node->valid.load(std::memory_order_acquire) && 
                node->key == key) {
                return node->value.load(std::memory_order_acquire);
            }
            node = node->next.load(std::memory_order_acquire);
        }
        
        return std::nullopt;
    }
    
    void put(const K& key, const V& value) {
        size_t idx = hasher_(key) % BUCKET_COUNT;
        
        // Check if key exists
        Node* node = buckets_[idx].load(std::memory_order_acquire);
        while (node) {
            if (node->valid.load(std::memory_order_acquire) && 
                node->key == key) {
                node->value.store(value, std::memory_order_release);
                return;
            }
            node = node->next.load(std::memory_order_acquire);
        }
        
        // Insert new node at head
        Node* new_node = new Node(key, value);
        Node* head = buckets_[idx].load(std::memory_order_relaxed);
        
        do {
            new_node->next.store(head, std::memory_order_relaxed);
        } while (!buckets_[idx].compare_exchange_weak(
            head, new_node,
            std::memory_order_release,
            std::memory_order_relaxed));
        
        size_.fetch_add(1, std::memory_order_relaxed);
    }
    
    bool remove(const K& key) {
        size_t idx = hasher_(key) % BUCKET_COUNT;
        Node* node = buckets_[idx].load(std::memory_order_acquire);
        
        while (node) {
            if (node->key == key) {
                bool expected = true;
                if (node->valid.compare_exchange_strong(
                        expected, false, std::memory_order_acq_rel)) {
                    size_.fetch_sub(1, std::memory_order_relaxed);
                    return true;
                }
                return false;
            }
            node = node->next.load(std::memory_order_acquire);
        }
        
        return false;
    }
    
    size_t size() const {
        return size_.load(std::memory_order_relaxed);
    }
};
```

### Approach 4: Read-Copy-Update (RCU)

For read-heavy workloads:

```cpp
template<typename K, typename V>
class RCUCache {
private:
    using Map = std::unordered_map<K, V>;
    
    std::atomic<Map*> current_;
    std::mutex write_mutex_;
    std::vector<Map*> old_maps_;  // For deferred deletion
    
public:
    RCUCache() : current_(new Map()) {}
    
    ~RCUCache() {
        delete current_.load();
        for (auto* m : old_maps_) {
            delete m;
        }
    }
    
    // Read - completely lock-free!
    std::optional<V> get(const K& key) {
        Map* map = current_.load(std::memory_order_acquire);
        auto it = map->find(key);
        if (it != map->end()) {
            return it->second;
        }
        return std::nullopt;
    }
    
    // Write - creates a copy
    void put(const K& key, const V& value) {
        std::lock_guard<std::mutex> lock(write_mutex_);
        
        Map* old_map = current_.load(std::memory_order_relaxed);
        Map* new_map = new Map(*old_map);  // Copy
        
        (*new_map)[key] = value;
        
        current_.store(new_map, std::memory_order_release);
        
        // Defer deletion (in practice, use epoch-based reclamation)
        old_maps_.push_back(old_map);
        
        // Simple cleanup: delete old maps after accumulating many
        if (old_maps_.size() > 100) {
            // In production, use proper synchronization
            for (auto* m : old_maps_) {
                delete m;
            }
            old_maps_.clear();
        }
    }
};
```

```cpp
template<typename K, typename V>
class MemoryLimitedCache {
public:
    using SizeFunc = std::function<size_t(const K&, const V&)>;
    
private:
    struct Entry {
        K key;
        V value;
        size_t memory_size;
        Entry* prev = nullptr;
        Entry* next = nullptr;
    };
    
    std::unordered_map<K, Entry*> cache_;
    Entry head_, tail_;
    
    size_t max_memory_;
    size_t current_memory_ = 0;
    SizeFunc size_func_;
    
    void evictUntilFits(size_t required_size) {
        while (current_memory_ + required_size > max_memory_ && !isEmpty()) {
            evictLRU();
        }
    }
    
    void evictLRU() {
        Entry* lru = tail_.prev;
        if (lru == &head_) return;
        
        removeFromList(lru);
        current_memory_ -= lru->memory_size;
        cache_.erase(lru->key);
        delete lru;
    }
    
    void removeFromList(Entry* e) {
        e->prev->next = e->next;
        e->next->prev = e->prev;
    }
    
    void addToFront(Entry* e) {
        e->next = head_.next;
        e->prev = &head_;
        head_.next->prev = e;
        head_.next = e;
    }
    
    bool isEmpty() const {
        return head_.next == &tail_;
    }
    
public:
    MemoryLimitedCache(size_t max_memory_bytes, SizeFunc size_func = nullptr)
        : max_memory_(max_memory_bytes)
        , size_func_(size_func ? size_func : defaultSizeFunc) {
        head_.next = &tail_;
        tail_.prev = &head_;
    }
    
    ~MemoryLimitedCache() {
        for (auto& [key, entry] : cache_) {
            delete entry;
        }
    }
    
    static size_t defaultSizeFunc(const K& k, const V& v) {
        // Approximate size for common types
        size_t size = sizeof(Entry);
        
        if constexpr (std::is_same_v<K, std::string>) {
            size += k.capacity();
        }
        if constexpr (std::is_same_v<V, std::string>) {
            size += v.capacity();
        }
        
        return size;
    }
    
    void put(const K& key, const V& value) {
        size_t entry_size = size_func_(key, value);
        
        // Remove old entry if exists
        auto it = cache_.find(key);
        if (it != cache_.end()) {
            current_memory_ -= it->second->memory_size;
            removeFromList(it->second);
            delete it->second;
            cache_.erase(it);
        }
        
        // Evict until we have space
        evictUntilFits(entry_size);
        
        // Check if single entry is too large
        if (entry_size > max_memory_) {
            return;  // Entry doesn't fit even in empty cache
        }
        
        // Insert new entry
        Entry* entry = new Entry{key, value, entry_size};
        cache_[key] = entry;
        addToFront(entry);
        current_memory_ += entry_size;
    }
    
    std::optional<V> get(const K& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return std::nullopt;
        }
        
        // Move to front
        removeFromList(it->second);
        addToFront(it->second);
        
        return it->second->value;
    }
    
    size_t memoryUsage() const { return current_memory_; }
    size_t maxMemory() const { return max_memory_; }
    float memoryUtilization() const {
        return static_cast<float>(current_memory_) / max_memory_;
    }
};
```

### Approach 3: Custom Allocator with Pool

Control memory allocation precisely:

```cpp
template<typename T>
class PoolAllocator {
private:
    struct Block {
        alignas(T) char data[sizeof(T)];
        Block* next = nullptr;
    };
    
    std::vector<std::unique_ptr<Block[]>> chunks_;
    Block* free_list_ = nullptr;
    size_t chunk_size_;
    size_t allocated_ = 0;
    size_t max_items_;
    std::mutex mutex_;
    
public:
    PoolAllocator(size_t max_items, size_t chunk_size = 1024)
        : chunk_size_(chunk_size), max_items_(max_items) {
        allocateChunk();
    }
    
    T* allocate() {
        std::lock_guard<std::mutex> lock(mutex_);
        
        if (free_list_ == nullptr) {
            if (allocated_ >= max_items_) {
                return nullptr;  // Limit reached
            }
            allocateChunk();
        }
        
        Block* block = free_list_;
        free_list_ = block->next;
        allocated_++;
        
        return reinterpret_cast<T*>(block->data);
    }
    
    void deallocate(T* ptr) {
        std::lock_guard<std::mutex> lock(mutex_);
        
        Block* block = reinterpret_cast<Block*>(ptr);
        block->next = free_list_;
        free_list_ = block;
        allocated_--;
    }
    
    size_t allocated() const { return allocated_; }
    size_t capacity() const { return max_items_; }
    
private:
    void allocateChunk() {
        size_t size = std::min(chunk_size_, max_items_ - allocated_);
        if (size == 0) return;
        
        auto chunk = std::make_unique<Block[]>(size);
        
        // Link all blocks in chunk to free list
        for (size_t i = 0; i < size; ++i) {
            chunk[i].next = free_list_;
            free_list_ = &chunk[i];
        }
        
        chunks_.push_back(std::move(chunk));
    }
};

// Usage with cache
template<typename K, typename V>
class PoolBackedCache {
private:
    struct Node {
        K key;
        V value;
        Node* prev;
        Node* next;
    };
    
    PoolAllocator<Node> allocator_;
    std::unordered_map<K, Node*> cache_;
    Node head_, tail_;
    
public:
    PoolBackedCache(size_t max_entries) 
        : allocator_(max_entries) {
        head_.next = &tail_;
        tail_.prev = &head_;
    }
    
    void put(const K& key, const V& value) {
        auto it = cache_.find(key);
        if (it != cache_.end()) {
            it->second->value = value;
            moveToFront(it->second);
            return;
        }
        
        Node* node = allocator_.allocate();
        if (node == nullptr) {
            // Pool exhausted, evict LRU
            evictLRU();
            node = allocator_.allocate();
        }
        
        new (node) Node{key, value, nullptr, nullptr};
        cache_[key] = node;
        addToFront(node);
    }
```

### Memory Monitoring

```cpp
class MemoryMonitor {
private:
    std::atomic<size_t> current_usage_{0};
    size_t max_memory_;
    size_t high_water_mark_;
    size_t low_water_mark_;
    std::function<void()> eviction_callback_;
    
public:
    MemoryMonitor(size_t max_memory, 
                  float high_water = 0.9,
                  float low_water = 0.7)
        : max_memory_(max_memory)
        , high_water_mark_(static_cast<size_t>(max_memory * high_water))
        , low_water_mark_(static_cast<size_t>(max_memory * low_water)) {}
    
    void setEvictionCallback(std::function<void()> callback) {
        eviction_callback_ = std::move(callback);
    }
    
    bool tryAllocate(size_t bytes) {
        size_t current = current_usage_.load(std::memory_order_relaxed);
        
        if (current + bytes > max_memory_) {
            return false;
        }
        
        while (!current_usage_.compare_exchange_weak(
                current, current + bytes,
                std::memory_order_relaxed)) {
            if (current + bytes > max_memory_) {
                return false;
            }
        }
        
        // Trigger eviction if above high water mark
        if (current + bytes > high_water_mark_ && eviction_callback_) {
            eviction_callback_();
        }
        
        return true;
    }
    
    void deallocate(size_t bytes) {
        current_usage_.fetch_sub(bytes, std::memory_order_relaxed);
    }
    
    size_t usage() const {
        return current_usage_.load(std::memory_order_relaxed);
    }
    
    float utilizationRatio() const {
        return static_cast<float>(usage()) / max_memory_;
    }
    
    bool isAboveHighWater() const {
        return usage() > high_water_mark_;
    }
    
    bool isBelowLowWater() const {
        return usage() < low_water_mark_;
    }
};
```

## Complete Implementation

A production-ready cache combining all concepts:

```cpp
#pragma once

#include <atomic>
#include <chrono>
#include <condition_variable>
#include <functional>
#include <list>
#include <memory>
#include <mutex>
#include <optional>
#include <shared_mutex>
#include <thread>
#include <unordered_map>
#include <vector>

namespace kvcache {

// ============================================================================
// Configuration
// ============================================================================

enum class EvictionPolicy {
    LRU,
    LFU,
    FIFO
};

struct CacheConfig {
    size_t max_memory_bytes = 100 * 1024 * 1024;  // 100 MB
    size_t max_entries = 0;  // 0 = unlimited (use memory limit)
    size_t shard_count = 16;
    EvictionPolicy eviction_policy = EvictionPolicy::LRU;
    std::chrono::milliseconds default_ttl{0};  // 0 = no expiry
    std::chrono::milliseconds cleanup_interval{1000};
    float high_water_mark = 0.9f;
    float low_water_mark = 0.7f;
};

// ============================================================================
// Statistics
// ============================================================================

struct CacheStats {
    std::atomic<uint64_t> hits{0};
    std::atomic<uint64_t> misses{0};
    std::atomic<uint64_t> evictions{0};
    std::atomic<uint64_t> expirations{0};
    std::atomic<size_t> current_size{0};
    std::atomic<size_t> current_memory{0};
    
    double hitRate() const {
        uint64_t total = hits.load() + misses.load();
        return total > 0 ? static_cast<double>(hits.load()) / total : 0.0;
    }
    
    void reset() {
        hits = 0;
        misses = 0;
        evictions = 0;
        expirations = 0;
    }
};

// ============================================================================
// Cache Entry
// ============================================================================

template<typename K, typename V>
struct CacheEntry {
    K key;
    V value;
    size_t memory_size;
    std::chrono::steady_clock::time_point expiry;
    std::chrono::steady_clock::time_point last_access;
    size_t access_count = 1;
    
    // For linked list
    CacheEntry* prev = nullptr;
    CacheEntry* next = nullptr;
    
    bool hasExpiry() const {
        return expiry != std::chrono::steady_clock::time_point{};
    }
    
    bool isExpired() const {
        return hasExpiry() && std::chrono::steady_clock::now() >= expiry;
    }
};

// ============================================================================
// Single Shard (LRU)
// ============================================================================

template<typename K, typename V>
class CacheShard {
public:
    using Entry = CacheEntry<K, V>;
    using SizeFunc = std::function<size_t(const K&, const V&)>;
    
private:
    std::unordered_map<K, Entry*> map_;
    Entry head_, tail_;  // Dummy nodes for LRU list
    
    size_t max_memory_;
    size_t current_memory_ = 0;
    size_t max_entries_;
    SizeFunc size_func_;
    CacheStats& stats_;
    std::chrono::milliseconds default_ttl_;
    
    mutable std::shared_mutex mutex_;
    
public:
    CacheShard(size_t max_memory, size_t max_entries,
               SizeFunc size_func, CacheStats& stats,
               std::chrono::milliseconds default_ttl)
        : max_memory_(max_memory)
        , max_entries_(max_entries)
        , size_func_(std::move(size_func))
        , stats_(stats)
        , default_ttl_(default_ttl) {
        head_.next = &tail_;
        tail_.prev = &head_;
    }
    
    ~CacheShard() {
        std::unique_lock lock(mutex_);
        for (auto& [key, entry] : map_) {
            delete entry;
        }
    }
    
    std::optional<V> get(const K& key) {
        std::unique_lock lock(mutex_);
        
        auto it = map_.find(key);
        if (it == map_.end()) {
            stats_.misses.fetch_add(1, std::memory_order_relaxed);
            return std::nullopt;
        }
        
        Entry* entry = it->second;
        
        // Check expiry
        if (entry->isExpired()) {
            removeEntry(entry);
            stats_.misses.fetch_add(1, std::memory_order_relaxed);
            stats_.expirations.fetch_add(1, std::memory_order_relaxed);
            return std::nullopt;
        }
        
        // Update access info
        entry->last_access = std::chrono::steady_clock::now();
        entry->access_count++;
        
        // Move to front (LRU)
        moveToFront(entry);
        
        stats_.hits.fetch_add(1, std::memory_order_relaxed);
        return entry->value;
    }
    
    void put(const K& key, const V& value,
             std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        std::unique_lock lock(mutex_);
        
        auto now = std::chrono::steady_clock::now();
        size_t entry_size = size_func_(key, value);
        
        // Check if key exists
        auto it = map_.find(key);
        if (it != map_.end()) {
            Entry* entry = it->second;
            current_memory_ -= entry->memory_size;
            
            entry->value = value;
            entry->memory_size = entry_size;
            entry->last_access = now;
            entry->access_count++;
            
            if (ttl || default_ttl_.count() > 0) {
                entry->expiry = now + ttl.value_or(default_ttl_);
            }
            
            current_memory_ += entry_size;
            moveToFront(entry);
            
            stats_.current_memory.store(current_memory_, 
                                         std::memory_order_relaxed);
            return;
        }
        
        // Evict if necessary
        evictIfNeeded(entry_size);
        
        // Create new entry
        Entry* entry = new Entry{
            key, value, entry_size,
            (ttl || default_ttl_.count() > 0) 
                ? now + ttl.value_or(default_ttl_)
                : std::chrono::steady_clock::time_point{},
            now, 1
        };
        
        map_[key] = entry;
        addToFront(entry);
        current_memory_ += entry_size;
        
        stats_.current_size.fetch_add(1, std::memory_order_relaxed);
        stats_.current_memory.store(current_memory_, std::memory_order_relaxed);
    }
    
    bool remove(const K& key) {
        std::unique_lock lock(mutex_);
        
        auto it = map_.find(key);
        if (it == map_.end()) {
            return false;
        }
        
        removeEntry(it->second);
        return true;
    }
    
    void cleanupExpired() {
        std::unique_lock lock(mutex_);
        
        auto now = std::chrono::steady_clock::now();
        
        // Start from tail (oldest)
        Entry* entry = tail_.prev;
        while (entry != &head_) {
            Entry* prev = entry->prev;
            
            if (entry->isExpired()) {
                removeEntry(entry);
                stats_.expirations.fetch_add(1, std::memory_order_relaxed);
            }
            
            entry = prev;
        }
    }
    
    void evictToMemory(size_t target_memory) {
        std::unique_lock lock(mutex_);
        
        while (current_memory_ > target_memory && tail_.prev != &head_) {
            evictLRU();
        }
    }
    
    size_t size() const {
        std::shared_lock lock(mutex_);
        return map_.size();
    }
    
    size_t memoryUsage() const {
        std::shared_lock lock(mutex_);
        return current_memory_;
    }
    
    void clear() {
        std::unique_lock lock(mutex_);
        
        for (auto& [key, entry] : map_) {
            delete entry;
        }
        map_.clear();
        head_.next = &tail_;
        tail_.prev = &head_;
        
        stats_.current_size.fetch_sub(map_.size(), std::memory_order_relaxed);
        current_memory_ = 0;
        stats_.current_memory.store(0, std::memory_order_relaxed);
    }
    
private:
    void addToFront(Entry* entry) {
        entry->next = head_.next;
        entry->prev = &head_;
        head_.next->prev = entry;
        head_.next = entry;
    }
    
    void removeFromList(Entry* entry) {
        entry->prev->next = entry->next;
        entry->next->prev = entry->prev;
    }
    
    void moveToFront(Entry* entry) {
        removeFromList(entry);
        addToFront(entry);
    }
    
    void removeEntry(Entry* entry) {
        removeFromList(entry);
        map_.erase(entry->key);
        current_memory_ -= entry->memory_size;
        delete entry;
        
        stats_.current_size.fetch_sub(1, std::memory_order_relaxed);
        stats_.current_memory.store(current_memory_, std::memory_order_relaxed);
    }
    
    void evictLRU() {
        Entry* lru = tail_.prev;
        if (lru == &head_) return;
        
        removeEntry(lru);
        stats_.evictions.fetch_add(1, std::memory_order_relaxed);
    }
    
    void evictIfNeeded(size_t required_size) {
        // Check entry limit
        if (max_entries_ > 0 && map_.size() >= max_entries_) {
            evictLRU();
        }
        
        // Check memory limit
        while (current_memory_ + required_size > max_memory_ && 
               tail_.prev != &head_) {
            evictLRU();
        }
    }
};

// ============================================================================
// Sharded Cache
// ============================================================================

template<typename K, typename V>
class Cache {
public:
    using SizeFunc = std::function<size_t(const K&, const V&)>;
    
private:
    std::vector<std::unique_ptr<CacheShard<K, V>>> shards_;
    CacheConfig config_;
    CacheStats stats_;
    std::hash<K> hasher_;
    SizeFunc size_func_;
    
    // Background cleanup
    std::thread cleanup_thread_;
    std::atomic<bool> running_{false};
    std::condition_variable cleanup_cv_;
    std::mutex cleanup_mutex_;
    
    CacheShard<K, V>& getShard(const K& key) {
        size_t idx = hasher_(key) % shards_.size();
        return *shards_[idx];
    }
    
public:
    explicit Cache(CacheConfig config = {}, SizeFunc size_func = nullptr)
        : config_(std::move(config))
        , size_func_(size_func ? size_func : defaultSizeFunc) {
        
        size_t per_shard_memory = config_.max_memory_bytes / config_.shard_count;
        size_t per_shard_entries = config_.max_entries > 0 
            ? config_.max_entries / config_.shard_count 
            : 0;
        
        for (size_t i = 0; i < config_.shard_count; ++i) {
            shards_.push_back(std::make_unique<CacheShard<K, V>>(
                per_shard_memory, per_shard_entries,
                size_func_, stats_, config_.default_ttl
            ));
        }
        
        startCleanupThread();
    }
    
    ~Cache() {
        stopCleanupThread();
    }
    
    // Default size function
    static size_t defaultSizeFunc(const K& key, const V& value) {
        size_t size = sizeof(CacheEntry<K, V>);
        
        if constexpr (std::is_same_v<K, std::string>) {
            size += key.capacity();
        }
        if constexpr (std::is_same_v<V, std::string>) {
            size += value.capacity();
        }
        if constexpr (std::is_same_v<V, std::vector<uint8_t>>) {
            size += value.capacity();
        }
        
        return size;
    }
    
    // Core operations
    std::optional<V> get(const K& key) {
        return getShard(key).get(key);
    }
    
    void put(const K& key, const V& value,
             std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        getShard(key).put(key, value, ttl);
    }
    
    bool remove(const K& key) {
        return getShard(key).remove(key);
    }
    
    bool contains(const K& key) {
        return get(key).has_value();
    }
    
    // Convenience: get or compute
    V getOrCompute(const K& key, std::function<V()> compute_func,
                   std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        auto cached = get(key);
        if (cached) {
            return *cached;
        }
        
        V value = compute_func();
        put(key, value, ttl);
        return value;
    }
    
    // Bulk operations
    std::unordered_map<K, V> getMulti(const std::vector<K>& keys) {
        std::unordered_map<K, V> results;
        for (const auto& key : keys) {
            auto value = get(key);
            if (value) {
                results[key] = *value;
            }
        }
        return results;
    }
    
    void putMulti(const std::unordered_map<K, V>& entries,
                  std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        for (const auto& [key, value] : entries) {
            put(key, value, ttl);
        }
    }
    
    // Statistics
    const CacheStats& stats() const { return stats_; }
    
    size_t size() const {
        size_t total = 0;
        for (const auto& shard : shards_) {
            total += shard->size();
        }
        return total;
    }
    
    size_t memoryUsage() const {
        size_t total = 0;
        for (const auto& shard : shards_) {
            total += shard->memoryUsage();
        }
        return total;
    }
    
    float memoryUtilization() const {
        return static_cast<float>(memoryUsage()) / config_.max_memory_bytes;
    }
    
    // Management
    void clear() {
        for (auto& shard : shards_) {
            shard->clear();
        }
    }
    
    void triggerEviction() {
        size_t target_memory = static_cast<size_t>(
            config_.max_memory_bytes * config_.low_water_mark);
        size_t per_shard = target_memory / shards_.size();
        
        for (auto& shard : shards_) {
            shard->evictToMemory(per_shard);
        }
    }
    
private:
    void startCleanupThread() {
        running_ = true;
        cleanup_thread_ = std::thread([this] {
            while (running_) {
                std::unique_lock lock(cleanup_mutex_);
                cleanup_cv_.wait_for(lock, config_.cleanup_interval,
                    [this] { return !running_.load(); });
                
                if (!running_) break;
                
                // Cleanup expired entries
                for (auto& shard : shards_) {
                    shard->cleanupExpired();
                }
                
                // Check memory pressure
                if (memoryUtilization() > config_.high_water_mark) {
                    triggerEviction();
                }
            }
        });
    }
    
    void stopCleanupThread() {
        running_ = false;
        cleanup_cv_.notify_all();
        if (cleanup_thread_.joinable()) {
            cleanup_thread_.join();
        }
    }
};
```

### 1. Write-Through and Write-Back

```cpp
template<typename K, typename V>
class WriteThroughCache {
private:
    Cache<K, V> cache_;
    std::function<void(const K&, const V&)> write_func_;
    std::function<std::optional<V>(const K&)> read_func_;
    
public:
    WriteThroughCache(CacheConfig config,
                      std::function<void(const K&, const V&)> write_func,
                      std::function<std::optional<V>(const K&)> read_func)
        : cache_(config)
        , write_func_(std::move(write_func))
        , read_func_(std::move(read_func)) {}
    
    std::optional<V> get(const K& key) {
        // Check cache first
        auto cached = cache_.get(key);
        if (cached) {
            return cached;
        }
        
        // Load from backing store
        auto value = read_func_(key);
        if (value) {
            cache_.put(key, *value);
        }
        
        return value;
    }
    
    void put(const K& key, const V& value) {
        // Write to backing store first (synchronous)
        write_func_(key, value);
        
        // Then update cache
        cache_.put(key, value);
    }
};

template<typename K, typename V>
class WriteBackCache {
private:
    struct DirtyEntry {
        K key;
        V value;
        std::chrono::steady_clock::time_point dirty_since;
    };
    
    Cache<K, V> cache_;
    std::function<void(const K&, const V&)> write_func_;
    std::function<std::optional<V>(const K&)> read_func_;
    
    std::queue<DirtyEntry> dirty_queue_;
    std::mutex dirty_mutex_;
    std::thread flush_thread_;
    std::atomic<bool> running_{true};
    std::chrono::milliseconds flush_interval_;
    
public:
    WriteBackCache(CacheConfig config,
                   std::function<void(const K&, const V&)> write_func,
                   std::function<std::optional<V>(const K&)> read_func,
                   std::chrono::milliseconds flush_interval = 
                       std::chrono::seconds(5))
        : cache_(config)
        , write_func_(std::move(write_func))
        , read_func_(std::move(read_func))
        , flush_interval_(flush_interval) {
        
        flush_thread_ = std::thread(&WriteBackCache::flushLoop, this);
    }
    
    ~WriteBackCache() {
        running_ = false;
        if (flush_thread_.joinable()) {
            flush_thread_.join();
        }
        flushAll();
    }
    
    void put(const K& key, const V& value) {
        // Update cache
        cache_.put(key, value);
        
        // Mark as dirty
        std::lock_guard lock(dirty_mutex_);
        dirty_queue_.push({key, value, std::chrono::steady_clock::now()});
    }
    
private:
    void flushLoop() {
        while (running_) {
            std::this_thread::sleep_for(flush_interval_);
            flushDirty();
        }
    }
    
    void flushDirty() {
        std::vector<DirtyEntry> to_flush;
        
        {
            std::lock_guard lock(dirty_mutex_);
            while (!dirty_queue_.empty()) {
                to_flush.push_back(std::move(dirty_queue_.front()));
                dirty_queue_.pop();
            }
        }
        
        for (const auto& entry : to_flush) {
            try {
                write_func_(entry.key, entry.value);
            } catch (...) {
                // Re-queue on failure
                std::lock_guard lock(dirty_mutex_);
                dirty_queue_.push(entry);
            }
        }
    }
    
    void flushAll() {
        flushDirty();
    }
};
```

### 2. Cache Warming

```cpp
template<typename K, typename V>
class WarmableCache {
private:
    Cache<K, V> cache_;
    
public:
    explicit WarmableCache(CacheConfig config) : cache_(config) {}
    
    // Warm cache from a data source
    template<typename Iterator>
    void warmFrom(Iterator begin, Iterator end) {
        for (auto it = begin; it != end; ++it) {
            cache_.put(it->first, it->second);
        }
    }
    
    // Warm from file
    void warmFromFile(const std::string& filename,
                      std::function<std::pair<K, V>(const std::string&)> parser) {
        std::ifstream file(filename);
        std::string line;
        
        while (std::getline(file, line)) {
            auto [key, value] = parser(line);
            cache_.put(key, value);
        }
    }
    
    // Async warming with progress callback
    void warmAsync(std::function<std::vector<std::pair<K, V>>()> loader,
                   std::function<void(size_t, size_t)> progress_callback = nullptr) {
        std::thread([this, loader, progress_callback] {
            auto entries = loader();
            size_t total = entries.size();
            
            for (size_t i = 0; i < entries.size(); ++i) {
                cache_.put(entries[i].first, entries[i].second);
                
                if (progress_callback && i % 1000 == 0) {
                    progress_callback(i, total);
                }
            }
            
            if (progress_callback) {
                progress_callback(total, total);
            }
        }).detach();
    }
};
```

### 3. Distributed Cache Support

```cpp
// Interface for distributed cache backends
template<typename K, typename V>
class DistributedCacheBackend {
public:
    virtual ~DistributedCacheBackend() = default;
    virtual std::optional<V> get(const K& key) = 0;
    virtual void put(const K& key, const V& value, 
                     std::chrono::milliseconds ttl) = 0;
    virtual bool remove(const K& key) = 0;
    virtual void invalidate(const K& key) = 0;  // Notify other nodes
};

template<typename K, typename V>
class TwoLevelCache {
private:
    Cache<K, V> local_cache_;
    std::unique_ptr<DistributedCacheBackend<K, V>> remote_cache_;
    
public:
    TwoLevelCache(CacheConfig local_config,
                  std::unique_ptr<DistributedCacheBackend<K, V>> remote)
        : local_cache_(local_config)
        , remote_cache_(std::move(remote)) {}
    
    std::optional<V> get(const K& key) {
        // Check local first
        auto local = local_cache_.get(key);
        if (local) {
            return local;
        }
        
        // Check remote
        auto remote = remote_cache_->get(key);
        if (remote) {
            // Populate local cache
            local_cache_.put(key, *remote);
        }
        
        return remote;
    }
    
    void put(const K& key, const V& value,
             std::optional<std::chrono::milliseconds> ttl = std::nullopt) {
        // Write to both
        local_cache_.put(key, value, ttl);
        remote_cache_->put(key, value, ttl.value_or(std::chrono::minutes(5)));
    }
    
    void invalidate(const K& key) {
        local_cache_.remove(key);
        remote_cache_->invalidate(key);  // Notify other nodes
    }
};
```
